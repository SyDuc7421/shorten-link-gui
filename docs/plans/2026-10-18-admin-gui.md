# Admin GUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the admin area of `shorten-link-gui` — dashboard, users, user detail, links and audit log under `/[locale]/admin/*` — against the admin API that is already merged in `shorten-link`.

**Architecture:** A new `src/features/admin/` feature in the repo's MVVM layout. `model/` holds Zod schemas (every ID a string), TanStack Query hooks over `authedHttp`, and pure helpers. `view-models/` holds one hook per screen plus three shared ones (URL filters, list toolbar, confirm dialog). `components/` render only. Route files under `src/app/[locale]/admin/` parse `params`/`searchParams` with Zod and compose feature components; `admin/layout.tsx` wraps everything in `AdminGate`. Filters live in the URL; the page re-renders with the new `searchParams` and the query key follows.

**Tech Stack:** Next.js 16 App Router, TypeScript strict, TanStack Query v5, Zod v4, next-intl v4, shadcn/ui (new: `table`, `chart`, `select`, `textarea`), recharts (new, via `chart`), sonner, react-hook-form, Vitest, Playwright.

**Spec:** `docs/specs/2026-10-18-admin-design.md` (§6 GUI, §7 GUI testing). API contract: `../shorten-link/api/openapi.yaml`, `/api/admin/*`.

## Global Constraints

- Every rule in `CLAUDE.md` and `DESIGN.md` applies; `DESIGN.md` §20.1 self-check before each UI commit.
- IDs in admin JSON are strings (`/^\d+$/`). Never pass one through `Number()` — snowflakes exceed 2^53.
- Times are RFC 3339 UTC; dashboard `date` is `YYYY-MM-DD` in UTC — format with `timeZone: 'UTC'`.
- Lists: newest first, keyset cursor, 20 per page, `next_cursor` empty string on the last page.
- Write bodies take an optional `{"reason"}`, max 500 **bytes**.
- `quota_daily` range `0..100000`, integer.
- Stats `days` range offered in the UI: 7 / 30 / 90 (API accepts 1..90, default 30).
- Errors: `401` no/invalid token, `403 {"error":"forbidden"}` not an enabled admin, `404` unknown user/link, `400` bad input, `409 {"error":"cannot disable yourself"}`, `413` body too large.
- Disable, enable and delete always ask for confirmation; disable and delete offer a reason field. Result shown with a toast; the row updates from the mutation response.
- Search inputs debounce 300 ms and live in the URL (`?q=&status=`).
- `recharts` (pulled in by `pnpm ui:add chart`) is the **only** new npm dependency.
- Every string in `messages/en.json`; `pnpm i18n:check` passes.
- Named exports only, except Next.js route files. No `any`, no `as` casts, no `!`.
- Conventional Commits; never `--no-verify`. `pnpm verify` passes before the last commit.

## Review Focus

1. **Snowflake IDs round-trip exactly.** `/admin/users/7311234567890123457` must load user `…457`, not `…456` — Task 2 schema test keeps the exact string; the e2e fixture uses 19-digit IDs.
2. **An admin tries to disable themselves.** The button is hidden on their own detail page with a one-line explanation, and a `409` (other tab, race) shows "You cannot disable your own account." — Task 8 view-model test.
3. **Typing in search while the previous debounced write is still landing.** The input must not jump back to the older value; a sidebar link or the back button _must_ replace the box — Task 4 toolbar view-model tests.
4. **A reason of 251 accented letters (502 bytes).** Rejected in the dialog with a clear message instead of a `413` — Task 4 confirm view-model test.
5. **A background refetch fails after a mutation.** The rows already on screen stay; the list does not blank into an error state — Task 4 `toPagedListState` test.

```ts
    expect(toAdminFailure(new TypeError('Failed to fetch'))).toBe('network');
```

```ts
import { z } from 'zod';

/** Snowflake IDs exceed 2^53, so the admin API sends every ID as a string. */
export const idSchema = z.string().regex(/^\d+$/);

const nullableTime = z.string().nullable();

export const USER_STATUSES = ['active', 'disabled'] as const;
export const LINK_STATUSES = ['active', 'disabled', 'expired'] as const;
export const RESOURCE_TYPES = ['user', 'link'] as const;
/** Ranges the dashboard offers. The API accepts 1–90 days. */
export const STATS_RANGES = ['7', '30', '90'] as const;
export const DEFAULT_STATS_RANGE = '30';

/** Mirror the API: PATCH /api/admin/users/{id} and the optional reason body. */
export const QUOTA_MIN = 0;
export const QUOTA_MAX = 100_000;
export const REASON_MAX_BYTES = 500;

/** The API limits the reason in bytes; accented letters and emoji take 2–4. */
export function isReasonWithinMax(reason: string) {
  return new TextEncoder().encode(reason).length <= REASON_MAX_BYTES;
}

export const adminMeSchema = z.object({
  id: idSchema,
  email: z.string(),
  role: z.literal('admin'),
});

export const adminUserSchema = z.object({
  id: idSchema,
  email: z.string(),
  role: z.enum(['user', 'admin']),
  quota_daily: z.number().int(),
  disabled_at: nullableTime,
  created_at: z.string(),
  link_count: z.number().int(),
});

export const adminUserDetailSchema = adminUserSchema.extend({
  /** Links created since 00:00 UTC. */
  links_today: z.number().int(),
  total_clicks: z.number().int(),
});

export const userPageSchema = z.object({
  users: z.array(adminUserSchema),
  /** Empty string on the last page. */
  next_cursor: z.string().nullish(),
});

export const adminLinkSchema = z.object({
  id: idSchema,
  code: z.string(),
  short_url: z.string(),
  original_url: z.string(),
  clicks: z.number().int(),
  owner: z.object({ id: idSchema, email: z.string(), disabled_at: nullableTime }),
  disabled_at: nullableTime,
  expires_at: nullableTime,
  created_at: z.string(),
});

export const linkPageSchema = z.object({
  links: z.array(adminLinkSchema),
  next_cursor: z.string().nullish(),
});

export const auditEntrySchema = z.object({
  id: idSchema,
  /** `null` for entries written by the `server admin` command. */
  actor: z.object({ id: idSchema, email: z.string() }).nullable(),
  // Open strings: an action or resource type added to the API later still renders.
  action: z.string(),
  resource_type: z.string(),
  resource_id: z.string(),
  metadata: z.record(z.string(), z.unknown()).nullish(),
  created_at: z.string(),
});

export const auditPageSchema = z.object({
  entries: z.array(auditEntrySchema),
  next_cursor: z.string().nullish(),
});

const count = z.number().int();

export const dailyStatSchema = z.object({
  /** `YYYY-MM-DD`, a UTC day. */
  date: z.string(),
  new_users: count,
  new_links: count,
  clicks: count,
});

export const adminStatsSchema = z.object({
  totals: z.object({
    users: count,
    links: count,
    clicks: count,
    disabled_users: count,
    disabled_links: count,
  }),
  /** One entry per day, oldest first, zeros filled in. */
  daily: z.array(dailyStatSchema),
  /** Top 10 by clicks within the range. */
  top_links: z.array(z.object({ code: z.string(), clicks: count })),
});

// URL search params: anything unexpected falls back to "no filter".
const searchText = z.string().trim().max(200).catch('');

export const userFiltersSchema = z.object({
  q: searchText,
  status: z.enum(USER_STATUSES).optional().catch(undefined),
});

export const linkFiltersSchema = z.object({
  q: searchText,
  status: z.enum(LINK_STATUSES).optional().catch(undefined),
});

export const auditFiltersSchema = z.object({
  resource_type: z.enum(RESOURCE_TYPES).optional().catch(undefined),
  actor_id: idSchema.optional().catch(undefined),
});

export const statsFiltersSchema = z.object({
  days: z.enum(STATS_RANGES).catch(DEFAULT_STATS_RANGE),
});

export type AdminMe = z.infer<typeof adminMeSchema>;
export type AdminUser = z.infer<typeof adminUserSchema>;
export type AdminUserDetail = z.infer<typeof adminUserDetailSchema>;
export type UserPage = z.infer<typeof userPageSchema>;
export type AdminLink = z.infer<typeof adminLinkSchema>;
export type LinkPage = z.infer<typeof linkPageSchema>;
export type AuditEntry = z.infer<typeof auditEntrySchema>;
export type AuditPage = z.infer<typeof auditPageSchema>;
export type AdminStats = z.infer<typeof adminStatsSchema>;
export type DailyStat = z.infer<typeof dailyStatSchema>;
export type UserFilters = z.infer<typeof userFiltersSchema>;
export type LinkFilters = z.infer<typeof linkFiltersSchema>;
export type AuditFilters = z.infer<typeof auditFiltersSchema>;
export type StatsRange = (typeof STATS_RANGES)[number];

/** Link list filters, optionally narrowed to one owner (user detail page). */
export type LinkQuery = LinkFilters & { userId?: string };
```

```ts
import { HttpError } from '@/lib/http';

/** Why an admin request failed — each one has a message in `admin.errors`. */
export type AdminFailure =
  'selfDisable' | 'notFound' | 'forbidden' | 'invalid' | 'reasonTooLong' | 'network' | 'server';

export function hasStatus(error: unknown, status: number) {
  return error instanceof HttpError && error.status === status;
}

export function toAdminFailure(error: unknown): AdminFailure {
  // fetch rejects with a TypeError when the request never reached the server.
  if (error instanceof TypeError) return 'network';
  if (!(error instanceof HttpError)) return 'server';

  switch (error.status) {
    // The only conflict an admin write returns.
    case 409:
      return 'selfDisable';
    case 404:
      return 'notFound';
    // Admin access was revoked after the page loaded.
    case 403:
      return 'forbidden';
    case 413:
      return 'reasonTooLong';
    case 400:
      return 'invalid';
    default:
      return 'server';
  }
}
```

```ts
export function withQuery(path: string, params: Record<string, string | undefined>) {
  const search = new URLSearchParams();
  for (const [key, value] of Object.entries(params)) {
    if (value) search.set(key, value);
  }
  const query = search.toString();
  return query ? `${path}?${query}` : path;
```

```ts
  return useQuery({
    queryKey: adminKeys.user(id),
    queryFn: async () => adminUserDetailSchema.parse(await authedHttp<unknown>(userPath(id))),
    // An unknown user is an answer, not a blip.
    retry: (failureCount, error) => !hasStatus(error, 404) && failureCount < 1,

function replaceUser(data: InfiniteData<UserPage> | undefined, user: AdminUserDetail) {
  if (!data) return data;
  return {
    ...data,
    pages: data.pages.map((page) => ({
      ...page,
      users: page.users.map((row) => (row.id === user.id ? { ...row, ...user } : row)),
    })),
  };
}

  const queryClient = useQueryClient();

    queryClient.setQueryData(adminKeys.user(user.id), user);
    queryClient.setQueriesData<InfiniteData<UserPage>>(
      { queryKey: adminKeys.userLists() },
      (data) => replaceUser(data, user),
    );
    void invalidateAdminData(queryClient);
  };
}
```

```ts
import {

  linkPageSchema,
  type AdminLink,
  type LinkPage,
  type LinkQuery,

function linkPath(code: string) {
  return `/api/admin/links/${encodeURIComponent(code)}`;
}

export function useAdminLinks(query: LinkQuery) {
  return useInfiniteQuery({
    queryKey: adminKeys.linkList(query),
    queryFn: async ({ pageParam }) =>
      linkPageSchema.parse(
        await authedHttp<unknown>(
          withQuery('/api/admin/links', {
            q: query.q,
            status: query.status,
            user_id: query.userId,
            cursor: pageParam,
          }),
        ),
      ),
    initialPageParam: '',
    getNextPageParam: (lastPage) => lastPage.next_cursor || undefined,
    placeholderData: keepPreviousData,
  });
}

function mapLinks(
  data: InfiniteData<LinkPage> | undefined,
  map: (links: AdminLink[]) => AdminLink[],
) {
  if (!data) return data;
  return { ...data, pages: data.pages.map((page) => ({ ...page, links: map(page.links) })) };
}

export function useSetLinkDisabled() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({
      code,
      isDisabled,
      reason,
    }: {
      code: string;
      isDisabled: boolean;
      reason: string | null;
    }) =>
      adminLinkSchema.parse(
        await authedHttp<unknown>(`${linkPath(code)}/${isDisabled ? 'disable' : 'enable'}`, {
          method: 'POST',
          body: withReason(reason),
        }),
      ),
    onSuccess: (link) => {
      queryClient.setQueriesData<InfiniteData<LinkPage>>(
        { queryKey: adminKeys.linkLists() },
        (data) =>
          mapLinks(data, (links) => links.map((row) => (row.code === link.code ? link : row))),
      );
      // The row may now belong to another status filter; the lists refetch.
      void invalidateAdminData(queryClient);
    },
  });
}

export function useDeleteAdminLink() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: async ({ code, reason }: { code: string; reason: string | null }) => {
      try {
        await authedHttp<null>(linkPath(code), { method: 'DELETE', body: withReason(reason) });
      } catch (error) {
        // Already gone is the outcome the admin asked for.
        if (!hasStatus(error, 404)) throw error;
      }
    },
    onSuccess: (_data, { code }) => {
      queryClient.setQueriesData<InfiniteData<LinkPage>>(
        { queryKey: adminKeys.linkLists() },
        (data) => mapLinks(data, (links) => links.filter((row) => row.code !== code)),
      );
      void invalidateAdminData(queryClient);
    },
  });
}
```

```ts
'use client';

import { useCallback } from 'react';
import { useSearchParams } from 'next/navigation';

import { usePathname, useRouter } from '@/i18n/navigation';

/**
 * Writes list filters to the URL so a view survives reload and can be shared
 * (DESIGN.md §7). The page re-renders with the new `searchParams`.
 */
export function useUrlFilters() {
  const router = useRouter();
  const pathname = usePathname();
  const searchParams = useSearchParams();

  // A dependency of the debounced search effect, so its identity matters.
  return useCallback(
    (patch: Record<string, string | undefined>) => {
      const next = new URLSearchParams(searchParams.toString());
      for (const [key, value] of Object.entries(patch)) {
        if (value) next.set(key, value);
        else next.delete(key);
      }
      const query = next.toString();
      router.replace(query ? `${pathname}?${query}` : pathname, { scroll: false });
    },
    [router, pathname, searchParams],
  );
}
```

```ts
'use client';

import { useEffect, useState } from 'react';

import { useUrlFilters } from './use-url-filters';

/** Wait for a pause in typing before the URL — and so the query — changes. */
export const SEARCH_DELAY_MS = 300;
/** Select value for "no filter"; Radix Select cannot use an empty string. */
export const ALL_OPTION = 'all';

interface ListFilters {
  q: string;
  status?: string;
}

export function useListToolbarViewModel(filters: ListFilters) {
  const setFilters = useUrlFilters();
  const [search, setSearch] = useState(filters.q);
  /** The last search this box wrote to the URL. */
  const [requested, setRequested] = useState(filters.q);
  const [seenQuery, setSeenQuery] = useState(filters.q);

  // The URL changed. Our own debounced write coming back is not news — the
  // admin may have typed more since. Anything else (sidebar link, back
  // button) replaces what the box shows.
  if (filters.q !== seenQuery) {
    setSeenQuery(filters.q);
    if (filters.q !== requested) {
      setSearch(filters.q);
      setRequested(filters.q);
    }
  }

  useEffect(() => {
    const next = search.trim();
    if (next === filters.q) return;
    const timeout = window.setTimeout(() => {
      setRequested(next);
      setFilters({ q: next || undefined });
    }, SEARCH_DELAY_MS);
    return () => window.clearTimeout(timeout);
  }, [search, filters.q, setFilters]);

  return {
    search,
    onSearchChange: setSearch,
    status: filters.status ?? ALL_OPTION,
    onStatusChange: (status: string) =>
      setFilters({ status: status === ALL_OPTION ? undefined : status }),
    hasFilters: filters.q !== '' || filters.status !== undefined,
    clearFilters: () => {
      setSearch('');
      setRequested('');
      setFilters({ q: undefined, status: undefined });
    },
  };
}

export type ListToolbarModel = ReturnType<typeof useListToolbarViewModel>;
```

```ts
import type { InfiniteData } from '@tanstack/react-query';

export type PagedListState<T> =
  | { status: 'pending' }
  | { status: 'error'; retry: () => void }
  | { status: 'empty' }
  | {
      status: 'success';
      items: T[];
      hasNextPage: boolean;
      isFetchingNextPage: boolean;
      /** A background refetch or a new filter, not a next page — the view dims the rows. */
      isRefreshing: boolean;
      loadMore: () => void;
    };

interface PagedQuery<Page> {
  data: InfiniteData<Page> | undefined;
  status: 'pending' | 'error' | 'success';
  hasNextPage: boolean;
  isFetchingNextPage: boolean;
  isRefetching: boolean;
  refetch: () => unknown;
  fetchNextPage: () => unknown;
}

/** One cursor-paged list (users, links, audit) as the four states its view renders. */
export function toPagedListState<Page, T>(
  query: PagedQuery<Page>,
  select: (page: Page) => T[],
): PagedListState<T> {
  if (query.status === 'pending') return { status: 'pending' };
  // A failed background refetch keeps the rows that are already on screen.
  if (!query.data) return { status: 'error', retry: () => void query.refetch() };

  const items = query.data.pages.flatMap(select);
  if (items.length === 0) return { status: 'empty' };

  return {
    status: 'success',
    items,
    hasNextPage: query.hasNextPage,
    isFetchingNextPage: query.isFetchingNextPage,
    isRefreshing: query.isRefetching && !query.isFetchingNextPage,
    loadMore: () => void query.fetchNextPage(),
  };
}
```

```ts
'use client';

import { useState } from 'react';
import { useTranslations } from 'next-intl';
import { toast } from 'sonner';

import { toAdminFailure, type AdminFailure } from '../model/admin-errors';
import { isReasonWithinMax } from '../model/types';

interface ConfirmActionOptions {
  /** Runs the request. A rejection keeps the dialog open and shows why. */
  onConfirm: (reason: string | null) => Promise<unknown>;
  /** Toasted once the request lands. */
  successMessage: string;
  /** After success — e.g. move focus off a row that is about to disappear. */
  onDone?: () => void;
}

export function useConfirmActionViewModel({
  onConfirm,
  successMessage,
  onDone,
}: ConfirmActionOptions) {
  const t = useTranslations('admin.errors');
  const [isOpen, setIsOpen] = useState(false);
  const [reason, setReason] = useState('');
  const [isPending, setIsPending] = useState(false);
  const [failure, setFailure] = useState<AdminFailure | null>(null);

  function onOpenChange(open: boolean) {
    // Closing mid-request would hide the outcome.
    if (isPending) return;
    if (!open) {
      setReason('');
      setFailure(null);
    }
    setIsOpen(open);
  }

  async function run() {
    const trimmed = reason.trim();
    if (!isReasonWithinMax(trimmed)) {
      setFailure('reasonTooLong');
      return;
    }

    setFailure(null);
    setIsPending(true);
    try {
      await onConfirm(trimmed || null);
      setIsOpen(false);
      setReason('');
      toast.success(successMessage);
      onDone?.();
    } catch (error) {
      setFailure(toAdminFailure(error));
    } finally {
      setIsPending(false);
    }
  }

  return {
    isOpen,
    onOpenChange,
    reason,
    onReasonChange: setReason,
    isPending,
    errorMessage: failure ? t(failure) : null,
    confirm: () => void run(),
  };
}
```

> `confirm` returns `void` (it is an `onClick`), so the tests wait for the settled state with `waitFor` instead of awaiting it.

```ts
'use client';

import { useEffect } from 'react';

import { useSession } from '@/features/auth';
import { useRouter } from '@/i18n/navigation';

import { hasStatus } from '../model/admin-errors';
import { useAdminMe } from '../model/me-api';

export type AdminGateState =
  | { status: 'pending' }
  | { status: 'forbidden' }
  | { status: 'error'; retry: () => void }
  | { status: 'allowed' };

/** UX only — the API checks the role on every /api/admin request. */
export function useAdminGateViewModel(): AdminGateState {
  const { status: sessionStatus } = useSession();
  const router = useRouter();
  const me = useAdminMe();

  useEffect(() => {
    if (sessionStatus === 'signedOut') router.replace('/login');
  }, [sessionStatus, router]);

  if (sessionStatus !== 'signedIn' || me.status === 'pending') return { status: 'pending' };
  if (me.status === 'error') {
    // Signed in but not an (enabled) admin: nothing failed (DESIGN.md §10).
    if (hasStatus(me.error, 403)) return { status: 'forbidden' };
    return { status: 'error', retry: () => void me.refetch() };
  }
  return { status: 'allowed' };
}
```

```ts
'use client';

import { useRef } from 'react';

import { getAdminLinkStatus, type AdminLinkStatus } from '../model/link-status';
import { useAdminLinks } from '../model/links-api';
import type { AdminLink, LinkQuery } from '../model/types';
import { toPagedListState } from './paged-list-state';
import { useListToolbarViewModel } from './use-list-toolbar-view-model';

export interface AdminLinkRow {
  link: AdminLink;
  status: AdminLinkStatus;
}

export function useLinksTableViewModel(query: LinkQuery) {
  const headingRef = useRef<HTMLHeadingElement>(null);
  const toolbar = useListToolbarViewModel(query);
  const now = new Date();
  const state = toPagedListState(useAdminLinks(query), (page) =>
    page.links.map((link): AdminLinkRow => ({ link, status: getAdminLinkStatus(link, now) })),
  );

  return {
    toolbar,
    state,
    headingRef,
    /** After a row is deleted, focus moves to something that still exists. */
    focusHeading: () => headingRef.current?.focus(),
  };
}
```

```ts
'use client';

import { hasStatus } from '../model/admin-errors';
import { useAdminMe } from '../model/me-api';
import type { AdminUserDetail } from '../model/types';
import { useAdminUser, useSetUserDisabled } from '../model/users-api';

export type UserDetailState =
  | { status: 'pending' }
  | { status: 'notFound' }
  | { status: 'error'; retry: () => void }
  | {
      status: 'success';
      user: AdminUserDetail;
      /** The API refuses self-disable (409); the view hides the action instead. */
      isSelf: boolean;
      isDisabled: boolean;
      toggleDisabled: (reason: string | null) => Promise<AdminUserDetail>;
    };

export function useUserDetailViewModel(id: string) {
  const query = useAdminUser(id);
  const me = useAdminMe();
  const setDisabled = useSetUserDisabled();

  let state: UserDetailState;
  if (query.status === 'pending') {
    state = { status: 'pending' };
  } else if (query.status === 'error') {
    state = hasStatus(query.error, 404)
      ? { status: 'notFound' }
      : { status: 'error', retry: () => void query.refetch() };
  } else {
    const user = query.data;
    const isDisabled = user.disabled_at !== null;
    state = {
      status: 'success',
      user,
      isSelf: me.data?.id === user.id,
      isDisabled,
      toggleDisabled: (reason) =>
        setDisabled.mutateAsync({ id: user.id, isDisabled: !isDisabled, reason }),
    };
  }

  return { state };
}
```

```ts
'use client';

import { zodResolver } from '@hookform/resolvers/zod';
import { useTranslations } from 'next-intl';
import { useForm } from 'react-hook-form';
import { toast } from 'sonner';
import { z } from 'zod';

import { toAdminFailure } from '../model/admin-errors';
import { isReasonWithinMax, QUOTA_MAX, QUOTA_MIN, type AdminUserDetail } from '../model/types';
import { useUpdateUserQuota } from '../model/users-api';

export function useQuotaFormViewModel(user: AdminUserDetail) {
  const t = useTranslations('admin.users');
  const tErrors = useTranslations('admin.errors');
  const mutation = useUpdateUserQuota();

  const quotaMessage = t('quotaInvalid', { min: QUOTA_MIN, max: QUOTA_MAX });
  // The input hands over a string; whole digits only, so "1.5" and "-1" fail here.
  const schema = z.object({
    quota: z
      .string()
      .trim()
      .regex(/^\d{1,6}$/, quotaMessage)
      .refine((value) => Number(value) <= QUOTA_MAX, quotaMessage),
    reason: z.string().trim().refine(isReasonWithinMax, tErrors('reasonTooLong')),
  });
  type QuotaValues = z.infer<typeof schema>;

  const form = useForm<QuotaValues>({
    resolver: zodResolver(schema),
    mode: 'onTouched',
    defaultValues: { quota: String(user.quota_daily), reason: '' },
  });

  function submit({ quota, reason }: QuotaValues) {
    form.clearErrors('root');
    mutation.mutate(
      { id: user.id, quotaDaily: Number(quota), reason: reason || null },
      {
        onSuccess: (updated) => {
          form.reset({ quota: String(updated.quota_daily), reason: '' });
          toast.success(t('quotaSaved'));
        },
        onError: (error) => form.setError('root', { message: tErrors(toAdminFailure(error)) }),
      },
    );
  }

  return {
    form,
    onSubmit: form.handleSubmit(submit),
    isPending: mutation.isPending,
    rootError: form.formState.errors.root?.message,
  };
}
```

> The test expects `100,000` — next-intl formats `{max, number}` with the `en` locale. If the zod `trim()` makes `'abc'`/`''` hit `regex` first, all five cases share the same message, which is what the test asserts.

```ts
'use client';

import { useAuditLog } from '../model/audit-api';
import {
  getAuditDetails,
  isKnownAuditAction,
  type AuditAction,
  type AuditDetails,
} from '../model/audit';
import type { AuditEntry, AuditFilters } from '../model/types';
import { toPagedListState } from './paged-list-state';
import { ALL_OPTION } from './use-list-toolbar-view-model';
import { useUrlFilters } from './use-url-filters';

export interface AuditRow {
  entry: AuditEntry;
  /** `null` for an action this GUI has no label for — the view shows it raw. */
  action: AuditAction | null;
  details: AuditDetails;
}

export function useAuditTableViewModel(filters: AuditFilters) {
  const setFilters = useUrlFilters();
  const state = toPagedListState(useAuditLog(filters), (page) =>
    page.entries.map((entry): AuditRow => ({
      entry,
      action: isKnownAuditAction(entry.action) ? entry.action : null,
      details: getAuditDetails(entry.metadata),
    })),
  );

  return {
    state,
    resourceType: filters.resource_type ?? ALL_OPTION,
    onResourceTypeChange: (value: string) =>
      setFilters({ resource_type: value === ALL_OPTION ? undefined : value }),
    isActorFiltered: filters.actor_id !== undefined,
    clearActor: () => setFilters({ actor_id: undefined }),
    hasFilters: filters.resource_type !== undefined || filters.actor_id !== undefined,
    clearFilters: () => setFilters({ resource_type: undefined, actor_id: undefined }),
  };
```

```ts
'use client';

import { sumDaily, type RangeTotals } from '../model/stats';
import { useAdminStats } from '../model/stats-api';
import { DEFAULT_STATS_RANGE, type AdminStats, type StatsRange } from '../model/types';
import { useUrlFilters } from './use-url-filters';

export type DashboardState =
  | { status: 'pending' }
  | { status: 'error'; retry: () => void }
  | { status: 'success'; stats: AdminStats; rangeTotals: RangeTotals; isRefreshing: boolean };

export function useAdminDashboardViewModel(days: StatsRange) {
  const setFilters = useUrlFilters();
  const { data, status, refetch, isPlaceholderData } = useAdminStats(days);

  let state: DashboardState;
  if (data) {
    state = {
      status: 'success',
      stats: data,
      rangeTotals: sumDaily(data.daily),
      isRefreshing: isPlaceholderData,
    };
  } else if (status === 'error') {
    state = { status: 'error', retry: () => void refetch() };
  } else {
    state = { status: 'pending' };
  }

  return {
    state,
    onDaysChange: (value: string) =>
      setFilters({ days: value === DEFAULT_STATS_RANGE ? undefined : value }),
  };
}
```
